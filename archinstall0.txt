# Archlinuxインストール備忘録メモ

## wifi接続
```
iwctl
    device list
    station wlan0 scan
    station wlan0 get-networks
    station wlan0 connect "SSID"
    exit

ping -c 3 archlinux.org
## 3 packets transmitted, 3 received, 0% packet loss 
```

## パーティション作成
```
### UEFIか確認
ls /sys/firmware/efi/
```

### パーティション手動作成
```
## デバイス確認
lsblk
```

```
## パーティションの削除 /dev/sdaの場合
sgdisk -Z /dev/sda

## パーティションの作成
sgdisk -n 1:0:+1G  -t 1:ef00  /dev/sda
sgdisk -n 2:0:+32G -t 2:8300  /dev/sda
sgdisk -n 3:0:0    -t 3:8300  /dev/sda
```

```
## 各パーティションをフォーマット
mkfs.fat -F 32 -n BOOT_DISK "/dev/sda1"
mkfs.ext4 -L ROOT_DISK "/dev/sda2"
mkfs.ext4 -L HOME_DISK "/dev/sda3"

lsblk -f /dev/sda
```

## 必須パッケージインストール
```
## pacstrap-install
mount -L ROOT_DISK /mnt/
mount --mkdir -L BOOT_DISK /mnt/boot
mount --mkdir -L HOME_DISK /mnt/home

pacman -Sy archlinux-keyring
pacstrap -K /mnt linux linux-firmware base base-devel linux-headers

## fstab
genfstab -U /mnt >> /mnt/etc/fstab
```


## chroot
```
arch-chroot /mnt/

    ## packages
    pacman -S neovim networkmanager

    ## タイムゾーン・時計
    ln -sf /usr/share/zoneinfo/Asia/Tokyo /etc/localtime
    hwclock --systohc

    ## ロケール
    vim /etc/locale.gen
        ## 以下のコメントアウトを解除。
        ja_JP.UTF-8 UTF-8
        en_US.UTF-8 UTF-8
    locale-gen
    localectl set-locale LANG=ja_JP.UTF-8

    ## ホスト名
    echo "HOSTNAME" > /etc/hostname
    
    ## root-pass
    passwd

    ## user-settings
    useradd -m -G wheel -s /bin/bash USERNAME
    passwd USERNAME

    ## sudo
    EDITOR=nvim visudo
        ## 以下のコメントアウトを解除。
        %wheel ALL=(ALL:ALL) ALL


    ## grub install(UEFIの場合)
    pacman -S grub dosfstools efibootmgr
    grub-install \
      --target=x86_64-efi \
      --efi-directory=/boot \
      --bootloader-id=grub \
      --recheck
    grub-mkconfig -o /boot/grub/grub.cfg

    ## reboot
    exit ## chrootから抜ける
reboot
```
